import java.util.Scanner;
/**
 *
 * @author fef
 */
public class Exercicios_jeferson {

    public static void main(String[] args) {
       Scanner ler = new Scanner(System.in);
       
        System.out.print("Digite S: ");
        int S = ler.nextInt();
        System.out.print("Digite A: ");
        int A = ler.nextInt();
        System.out.print("Digite B: ");
        int B = ler.nextInt();
        int quantidade = 0;

        for (int numero = A; numero <= B; numero++) {

            int valor = numero;
            int soma = 0;

            while (valor > 0) {
                soma += valor % 10;
                valor /= 10;
            } if (soma == S) {
                    quantidade++;
            }
        }

        System.out.println("Quantidade: " + quantidade);
    }
}

import java.util.Scanner;
/**
 *
 * @author fef
 */
public class Exercicios_jeferson {

    public static void main(String[] args) {
       Scanner ler = new Scanner(System.in);
       
        System.out.print("Digite o numero da primeira charrete: ");
        int N1 = ler.nextInt();
        System.out.print("Digite a distancia da primeira charrete: ");
        int D1 = ler.nextInt();
        System.out.print("Digite a velocidade da primeira charrete: ");
        int V1 = ler.nextInt();
        System.out.print("Digite o numero da segunda charrete: ");
        int N2 = ler.nextInt();

        System.out.print("Digite a distancia da segunda charrete: ");
        int D2 = ler.nextInt();

        System.out.print("Digite a velocidade da segunda charrete: ");
        int V2 = ler.nextInt();

        double tempo1 = (double) D1 / V1;
        double tempo2 = (double) D2 / V2;

        if (tempo1 < tempo2) {
            System.out.println("A charrete vencedora e: " + N1);
        } else {
            System.out.println("A charrete vencedora e: " + N2);
        }
    }
}
